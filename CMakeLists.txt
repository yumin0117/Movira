cmake_minimum_required(VERSION 3.20)
project(Movira VERSION 1.0 LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_EXPORT_COMPILE_COMMANDS ON)
if(NOT OpenCV_DIR AND EXISTS "${CMAKE_SOURCE_DIR}/.deps/opencv/build")
  set(OpenCV_DIR "${CMAKE_SOURCE_DIR}/.deps/opencv/build")
endif()
if(MSVC AND EXISTS "${CMAKE_SOURCE_DIR}/.deps/opencv/build/x64/vc16")
  set(OpenCV_ARCH x64)
  set(OpenCV_RUNTIME vc16)
endif()
find_package(OpenCV 4.10 REQUIRED COMPONENTS core imgproc imgcodecs videoio highgui dnn)
add_library(movira_pose src/pose.cpp)
target_include_directories(movira_pose PUBLIC src ${OpenCV_INCLUDE_DIRS})
target_link_libraries(movira_pose PUBLIC ${OpenCV_LIBS})
if(MSVC)
  target_compile_options(movira_pose PUBLIC /utf-8 /W4)
endif()
add_executable(Movira src/main.cpp)
target_link_libraries(Movira PRIVATE movira_pose)
enable_testing()
add_executable(movira_tests tests/pose_tests.cpp)
target_link_libraries(movira_tests PRIVATE movira_pose)
add_test(NAME pose_decoder COMMAND movira_tests)
if(WIN32 AND EXISTS "${OpenCV_DIR}/x64/vc16/bin")
  file(GLOB OPENCV_DLLS "${OpenCV_DIR}/x64/vc16/bin/opencv_world*.dll" "${OpenCV_DIR}/x64/vc16/bin/opencv_videoio_ffmpeg*.dll")
  foreach(target Movira movira_tests)
    add_custom_command(TARGET ${target} POST_BUILD COMMAND ${CMAKE_COMMAND} -E copy_if_different ${OPENCV_DLLS} $<TARGET_FILE_DIR:${target}> COMMAND_EXPAND_LISTS)
  endforeach()
endif()
